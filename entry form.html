Read Sales Data from the first active tab const dataSheet = ss.getSheets()[0]; const rows = dataSheet.getDataRange().getValues(); // Detect header indices dynamically const headers = rows[0].map(h => String(h).trim().toLowerCase()); let partyIdx = headers.indexOf("payer name"); let prodIdx = headers.indexOf("strength"); let qtyIdx = headers.indexOf("kg/ltrs"); // Fallbacks if header names have slight spacing variations if (partyIdx === -1) partyIdx = headers.findIndex(h => h.includes("payer name")); if (prodIdx === -1) prodIdx = headers.findIndex(h => h.includes("strength")); if (qtyIdx === -1) qtyIdx = headers.findIndex(h => h.includes("kg/ltrs")); // Hard defaults to SAP column layout (P, AC, AI) if (partyIdx === -1) partyIdx = 15; if (prodIdx === -1) prodIdx = 28; if (qtyIdx === -1) qtyIdx = 34; const billedMap = {}; for (let i = 1; i < rows.length; i++) { const row = rows[i]; const party = row[partyIdx] ?
